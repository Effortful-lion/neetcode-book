## 二分查找

### 核心题型

1. 精确查找target

2. 搜索包含target的边界

3. 找出target的近似值（最近的边界）

### 核心代码

```Go
// 左闭右闭（推荐）
func find1(target int, nums []int) int {
    left, right := 0, len(nums)-1
    for left &lt;= right {
       mid := left + (right-left)/2
       if nums[mid] == target {
          return mid
       } else if nums[mid] &gt; target {
          right = mid - 1
       }
       if nums[mid] &lt; target {
          left = mid + 1
       }
    }
    return -1
}
```

```Go
func FirstAndLast(nums []int, target int) []int {
    if len(nums) == 0 {
       return []int{-1, -1}
    }

    res_left, res_right := -1, -1
    left, right := 0, len(nums)-1

    // 先找left
    for left &lt;= right {
       mid := left + (right-left)/2
       if nums[mid] == target {
          res_left = mid
          right = mid - 1
       } else if nums[mid] &gt; target {
          right = mid - 1
       } else if nums[mid] &lt; target {
          left = mid + 1
       }
    }

    if res_left == -1 {
       return []int{-1, -1}
    }

    // 注意：重置 left 和 right
    left, right = 0, len(nums)-1
    for left &lt;= right {
       mid := left + (right-left)/2
       if nums[mid] == target {
          res_right = mid
          left = mid + 1
       } else if nums[mid] &gt; target {
          right = mid - 1
       } else if nums[mid] &lt; target {
          left = mid + 1
       }
    }

    return []int{res_left, res_right}
}
```

## 移除元素

### 核心题型

1. 双指针去某个target值\(指针可以是辅助，也可以是res\)

2. 含回退字符（字符串比较）

其实核心都是保留（找到）有效字符，忽略无效字符。

### 核心代码

```Go
func func1(nums []int, val int) int {
    // 双指针，一前一后覆盖，两个指针的距离就是删除的个数
    res := 0
    for i := 0; i &lt; len(nums); i++ {
       // 有效保留
       if nums[i] != val {
          nums[res] = nums[i]
          res++
       }
    }
    return res
}
```

```Go
func backspaceCompare(s string, t string) bool {
    p1, p2 := len(s)-1, len(t)-1
    count1, count2 := 0, 0

    for p1 &gt;= 0 || p2 &gt;= 0 {
       // 处理 s 的退格，找到下一个有效字符位置
       for p1 &gt;= 0 {
          if s[p1] == '#' {
             count1++
             p1--
          } else if count1 &gt; 0 {
             // 有退格计数，跳过当前字符（被删除）
             count1--
             p1--
          } else {
             // 找到有效字符，退出内层循环
             break
          }
       }

       // 处理 t 的退格，找到下一个有效字符位置
       for p2 &gt;= 0 {
          if t[p2] == '#' {
             count2++
             p2--
          } else if count2 &gt; 0 {
             // 有退格计数，跳过当前字符（被删除）
             count2--
             p2--
          } else {
             // 找到有效字符，退出内层循环
             break
          }
       }

       // 比较当前有效字符
       // 如果一个指针已越界，另一个未越界，说明不相等
       if (p1 &gt;= 0) != (p2 &gt;= 0) {
          return false
       }
       // 两个指针都未越界，但字符不同，说明不相等
       if p1 &gt;= 0 &amp;&amp; p2 &gt;= 0 &amp;&amp; s[p1] != t[p2] {
          return false
       }

       // 移动到下一个待处理位置
       p1--
       p2--
    }

    return true
}
```

```Go
func sortedSquares(nums []int) []int {
    // 没有要求空间复杂度
    res := make([]int, len(nums))
    left, right := 0, len(nums)-1
    count := len(nums) - 1
    for left &lt; right {
       tempLeft := nums[left] * nums[left]
       tempRight := nums[right] * nums[right]
       if tempLeft &gt;= tempRight {
          res[count] = tempLeft
          left++
       } else {
          res[count] = tempRight
          right--
       }
       count--
    }
    // left == right
    res[count] = nums[left] * nums[left]
    return res
}
```

## 长度最小的子数组

```Go
func minSubArrayLen(target int, nums []int) int {
    // 1. 返回数组最小长度
    // 2. 数组元素之和 &gt;= target
    left, right := 0, 0
    // 刚开始如果res=0，那么最小就是0，所以初始化为不可能值
    res := len(nums) + 1
    sum := nums[0]
    for right &lt; len(nums) {
        for sum &gt;= target {
            // 比较并记录
            cur := right - left + 1
            if cur &lt; res {
                res = cur
            }
            // 左指针向右缩小
            sum -= nums[left]
            left++
        }
        // 右指针继续向右扩大
        right++
        if right &lt; len(nums) {
            sum += nums[right]
        }
    }

    // 如果没有赋值，说明不存在。
    // 刚开始如果res=0，那么最小就是0，所以初始化为不可能值
    if res == len(nums)+1 {
        res = 0
    }
    return res
}
```

```Go
func totalFruit(fruits []int) int {
    // 哈希表记录当前窗口中两种水果的出现次数
    fruitCount := make(map[int]int)
    left := 0   // 窗口左边界
    maxLen := 0 // 最大长度结果

    // 右指针遍历所有水果
    for right, fruit := range fruits {
       // 将当前水果加入窗口
       fruitCount[fruit]++

       // 当窗口中水果种类超过2种时，移动左指针缩小窗口
       for len(fruitCount) &gt; 2 {
          leftFruit := fruits[left]
          fruitCount[leftFruit]-- // 减少左边界水果的计数
          if fruitCount[leftFruit] == 0 {
             delete(fruitCount, leftFruit) // 计数为0时移除该水果
          }
          left++ // 左指针右移
       }

       // 更新最大长度（当前窗口长度为 right - left + 1）
       currentLen := right - left + 1
       if currentLen &gt; maxLen {
          maxLen = currentLen
       }
    }

    return maxLen
}
```

## 螺旋矩阵

定义好边界，对边界进行控制。

螺旋矩阵一：矩阵元素输出

```Go
func spiralOrder(matrix [][]int) []int {
    if len(matrix) == 0 || len(matrix[0]) == 0 {
        return []int{}
    }
    Nrow, Ncol := len(matrix), len(matrix[0])
    n := Nrow * Ncol
    result := make([]int, 0, n)
    // 定义边界：上、下、左、右
    top, bottom := 0, Nrow-1
    left, right := 0, Ncol-1

    for len(result) &lt; n {
        // 从左到右遍历上边界
        for col := left; col &lt;= right; col++ {
            result = append(result, matrix[top][col])
        }
        top++ // 上边界下移
        if top &gt; bottom {
            break
        }

        // 从上到下遍历右边界
        for row := top; row &lt;= bottom; row++ {
            result = append(result, matrix[row][right])
        }
        right-- // 右边界左移
        if left &gt; right {
            break
        }

        // 从右到左遍历下边界
        for col := right; col &gt;= left; col-- {
            result = append(result, matrix[bottom][col])
        }
        bottom-- // 下边界上移
        if top &gt; bottom {
            break
        }

        // 从下到上遍历左边界
        for row := bottom; row &gt;= top; row-- {
            result = append(result, matrix[row][left])
        }
        left++ // 左边界右移
        if left &gt; right {
            break
        }
    }
    return result
}
```

```Go
func generateMatrix(n int) [][]int {

    if n &lt; 1{
        return [][]int{}
    }

    // 边界
    left, right,up,down := 0, n-1, 0, n-1
    // 循环次数
    loop := n/2
    // center(奇数就是中间，偶数不需要)
    // center := n/2

    // 计数器 1 - n方
    count := 1

    // 结果数组初始化
    res := make([][]int, n)
    for i:=0;i&lt;n;i++{
        res[i] = make([]int, n)
    }

    // 循环：控制边界
    for i:=0;i&lt;loop;i++{
        // 填充上边界
        for col:=left;col&lt;=right;col++{
            res[up][col] = count
            count++
        }
        // 填充完成，上边界下移
        up++
        if up &gt; down{
            break
        }

        // 填充右边界
        for row:=up;row&lt;=down;row++{
            res[row][right] = count
            count++
        }
        // 填充完成，右边界左移
        right--
        if right &lt; left{
            break
        }

        // 填充下边界
        for col:=right;col&gt;=left;col--{
            res[down][col] = count
            count++
        }
        // 填充完成，下边界上移
        down--
        if down &lt; up{
            break
        }

        // 填充左边界
        for row:=down;row&gt;=up;row--{
            res[row][left] = count
            count++
        }
        // 填充完成，左边界右移
        left++
        if left &gt; right{
            break
        }
    }

    // 判断 n 是否为 奇数
    // center(奇数就是中间，偶数不需要)
    center := n/2
    if n % 2 == 1{
        // 有中间元素
        res[center][center] = n * n // count
    }
    return res
}
```

## 区间和

简单的说：就是利用一次计算替代后面的重复计算。比如：我要某一个区间的和，可以每次都算一下；也可以先算每一个位置的前缀和，最后范围相减就得到了区间和了。当需要计算的次数为n，那么耗费的时间 m\(个数\) \+ n。否则，就需要n\*n

纯暴力有问题：


前缀区间解决（优化解法）

```Go
package main

import (
    "bufio"
    "fmt"
    "os"
    "strconv"
)

// 计算区间和：array 是前缀和数组
func calculateRangeSum(array []int, start, end int) int {
    // 前缀和
    prefixSumStart := 0
    // 如果不是第一个数，那么就初始化前缀和为：当前位置的前一个数字的前缀和
    if start &gt; 0 {
       prefixSumStart = array[start-1]
    }
    // 返回 end 的前缀和 - start 的前缀和
    return array[end] - prefixSumStart
}

func main() {

    // 读取性能优化 buffer scan
    scanner := bufio.NewScanner(os.Stdin)

    // 读取数组大小
    scanner.Scan()
    size, _ := strconv.Atoi(scanner.Text())

    // 读取数组元素
    numberArray := make([]int, size)
    for i := 0; i &lt; size; i++ {
       scanner.Scan()
       numberArray[i], _ = strconv.Atoi(scanner.Text())
    }

    // 计算前缀和数组
    prefixArray := make([]int, size)
    prefixArray[0] = numberArray[0]
    for i := 1; i &lt; size; i++ {
       prefixArray[i] = prefixArray[i-1] + numberArray[i]
    }

    // 计算区间和
    // scan.Scan()有返回值：表示是否有输入（对应直到没输入）
    var a, b int
    for scanner.Scan() {
       line := scanner.Text()
       // 根据输入格式，从 line 中获取对应位置的数字
       _, err := fmt.Sscanf(line, "%d %d", &amp;a, &amp;b)
       if err != nil {
          continue
       }

       if a &lt; 0 || b &gt;= size {
          continue
       }
       fmt.Println(calculateRangeSum(prefixArray, a, b))
    }
}
```

```Go
       }
    }

    // 计算最小差值
    fmt.Println(f(nums))
}

// f 函数：计算所有可能的单行划分或单列划分中，两部分和的最小差值
func f(nums [][]int) int {
    if len(nums) == 0 || len(nums[0]) == 0 {
       return 0
    }
    n, m := len(nums), len(nums[0])
    totalSum := 0 // 数组总和（用于快速计算划分后的另一部分和）

    // 计算每行的和 与 数组总和
    rowSums := make([]int, n)
    for i := 0; i &lt; n; i++ {
       for j := 0; j &lt; m; j++ {
          rowSums[i] += nums[i][j]
          totalSum += nums[i][j]
       }
    }

    // 计算每列的和
    colSums := make([]int, m)
    for j := 0; j &lt; m; j++ {
       for i := 0; i &lt; n; i++ {
          colSums[j] += nums[i][j]
       }
    }

    minDiff := math.MaxInt32 // 最小差值，初始化为最大整数

    // 1. 考虑所有可能的行划分（横向切一刀，分成上下两部分）
    // 划分位置：在第i行下方切（即上部分为0~i行，下部分为i+1~n-1行）
    upperSum := 0              // 上部分累加和
    for i := 0; i &lt; n-1; i++ { // 最多切n-1刀（最后一行下方无需切）
       upperSum += rowSums[i]
       lowerSum := totalSum - upperSum // 下部分和 = 总和 - 上部分和
       diff := abs(upperSum - lowerSum)
       if diff &lt; minDiff {
          minDiff = diff
       }
    }

    // 2. 考虑所有可能的列划分（纵向切一刀，分成左右两部分）
    // 划分位置：在第j列右方切（即左部分为0~j列，右部分为j+1~m-1列）
    leftSum := 0               // 左部分累加和
    for j := 0; j &lt; m-1; j++ { // 最多切m-1刀（最后一列右方无需切）
       leftSum += colSums[j]
       rightSum := totalSum - leftSum // 右部分和 = 总和 - 左部分和
       diff := abs(leftSum - rightSum)
       if diff &lt; minDiff {
          minDiff = diff
       }
    }

    return minDiff
}

func abs(x int) int {
    if x &lt; 0 {
       return -x
    }
    return x
}
```

```Go
func removeElements(head *ListNode, val int) *ListNode {
    if head == nil{
        return head

    // 注意：先递再归
    head.Next = removeElements(head.Next, val)
```

```Go
// 迭代
func removeElements(head *ListNode, val int) *ListNode {
    if head == nil{
        return nil
    }

    dummy:=&amp;ListNode{}
    dummy.Next = head

    temp := dummy
    for temp.Next != nil{
        next := temp.Next
        if next.Val == val{
            temp.Next = next.Next
            continue
        }
        temp = temp.Next
    }

    return dummy.Next
```

## 设计一个链表\(这里是一个双向链表\)

```Go
package main

import "fmt"

// 节点结构
type Node struct {
    Val  int
    Pre  *Node // 前驱指针
    Next *Node // 后继指针（最后一个节点为nil）
}

// 双向链表结构（带哨兵节点，非循环）
type MyLinkedList struct {
    sentinel *Node // 哨兵节点（仅作为头节点的前驱）
}

// 初始化链表：哨兵节点的next指向nil（空链表）
func Constructor() MyLinkedList {
    sentinel := &amp;Node{}
    sentinel.Next = nil // 哨兵的next初始为nil（无有效节点）
    sentinel.Pre = nil  // 哨兵无前驱
    return MyLinkedList{sentinel: sentinel}
}

// 获取索引为index的节点值
func (this *MyLinkedList) Get(index int) int {
    if index &lt; 0 {
        return -1
    }
    // 从第一个有效节点（哨兵的next）开始遍历
    curr := this.sentinel.Next
    for i := 0; i &lt; index; i++ {
        if curr == nil { // 索引超出范围
            return -1
        }
        curr = curr.Next
    }
    if curr == nil { // 索引超出范围
        return -1
    }
    return curr.Val
}

// 在头部插入节点
func (this *MyLinkedList) AddAtHead(val int) {
    newNode := &amp;Node{Val: val}
    // 新节点的next指向原头节点（哨兵的next）
    newNode.Next = this.sentinel.Next
    // 新节点的pre指向哨兵
    newNode.Pre = this.sentinel

    // 哨兵的next指向新节点（更新头节点）
    this.sentinel.Next = newNode

    // 若原头节点存在，其pre指向新节点
    if this.sentinel.Next != nil {
        this.sentinel.Next.Pre = newNode
    }
}

// 在尾部插入节点
func (this *MyLinkedList) AddAtTail(val int) {
    newNode := &amp;Node{Val: val}
    newNode.Next = nil // 尾节点的next必须为nil

    // 找到最后一个有效节点（next为nil的节点）
    curr := this.sentinel
    for curr.Next != nil {
        curr = curr.Next
    }

    // 最后一个节点的next指向新节点
    curr.Next = newNode
    // 新节点的pre指向最后一个节点
    newNode.Pre = curr
}

// 在索引index处插入节点
func (this *MyLinkedList) AddAtIndex(index int, val int) {
    if index &lt; 0 {
        this.AddAtHead(val)
        return
    }

    // 找到插入位置的前驱节点
    prev := this.sentinel
    for i := 0; i &lt; index; i++ {
        if prev.Next == nil { // 索引超出范围, 不插入
```

```Go
    sLen, pLen := len(s), len(p)
    if sLen &lt; pLen {

    // 创建统计表（26个字母就是26大小的数组）
    var sCount, pCount [26]int

        sCount[s[i+pLen]-'a']++

            ans = append(ans, i+1)

    }
```

```Go
import "sort"

func threeSum(nums []int) [][]int {
    sort.Ints(nums) // 先排序,双指针去重
    res := make([][]int, 0)
    n := len(nums)

    for i := 0; i &lt; n-2; i++ {
        // 跳过第一个元素的重复值
        if i &gt; 0 &amp;&amp; nums[i] == nums[i-1] {
            continue
        }
        left, right := i+1, n-1
        for left &lt; right {
            // 这里一定是先找到再去重
            // 去重是对已经找到的结果进行去重
            sum := nums[i] + nums[left] + nums[right]
            if sum == 0 {
                res = append(res, []int{nums[i], nums[left], nums[right]})
                // 跳过left和right的重复值
                for left &lt; right &amp;&amp; nums[left] == nums[left+1] {
                    left++
                }
                for left &lt; right &amp;&amp; nums[right] == nums[right-1] {
                    right--
                }
                left++
                right--
            } else if sum &lt; 0 {
                left++ // 和太小，左指针右移
            } else {
                right-- // 和太大，右指针左移
            }
        }
    }
    return res
}
```

```Go
func fourSum(nums []int, target int) [][]int {
    // 固定一个且去重，就是三数之和
    n := len(nums)

    // 先排序，为双指针提供基础
    sort.Ints(nums)

            // 第二个数固定且去重（j&gt;i+1 防止 出现开端到一直结束）
            if j&gt;i+1 &amp;&amp; nums[j]==nums[j-1]{

            for l&lt;r{
                sum := nums[i]+nums[j]+nums[l]+nums[r]

                    l++

                    l++
                }else{
                    r--
                }

    }
```

```Go
    // 注意： j:=0 必须要在for外面，因为j一直记录的是最长的前缀长度，不能每次都从0开始
    j:=0
    for i:=1;i&lt;len(str);i++{
        // i从1开始，因为一个字符没有前后缀，已经初始化了
        // 可能不是一步匹配
        for j&gt;0 &amp;&amp; str[j]!=str[i]{
            // j 需要回退：根据自身的prefix表，随时填充，随时使用的
            j=prefix[j-1]
        }    
        // str[j] == str[i] || j == 0
        if str[j] == str[i]{
            j++
        }
        prefix[i] = j
        // i++ for循环自然++
    }
```

```Go
func isValid(s string) bool {
    stack := make([]byte, 0) // 用切片模拟栈，存储左括号
    for i := 0; i &lt; len(s); i++ {
        c := s[i] // 直接按byte处理（括号均为单字节ASCII字符）
        if c == '(' || c == '[' || c == '{' {
            // 左括号入栈
            stack = append(stack, c)
        } else {
            // 右括号：检查栈是否为空或匹配
            if len(stack) == 0 {
                return false // 栈空却出现右括号，无效
            }
            // 弹出栈顶元素并比较
            top := stack[len(stack)-1]
            stack = stack[:len(stack)-1] // 弹出栈顶
            if top != match(c) {
                return false // 不匹配，无效
            }
        }
    }
    // 所有字符处理完后，栈必须为空才有效（无剩余左括号）
    return len(stack) == 0
```

```Go
func maxSlidingWindow(nums []int, k int) []int {
    if len(nums) == 0 || k &lt;= 0 || k &gt; len(nums) {
        return []int{}
    }

    res := make([]int, 0, len(nums)-k+1)
    deque := make([]int, 0) // 存储索引，确保对应的值单调递减

    for i := 0; i &lt; len(nums); i++ {
        // 1. 移除队列中超出当前窗口的索引（窗口范围：[i-k+1, i]）
        for len(deque) &gt; 0 &amp;&amp; deque[0] &lt; i-k+1 {
            deque = deque[1:] // 队首出队
        }

        // 2. 移除队列中比当前元素小的索引（保证队列单调递减）
        // 因为这些元素在当前元素之后，不可能成为窗口最大值
        for len(deque) &gt; 0 &amp;&amp; nums[i] &gt;= nums[deque[len(deque)-1]] {
            deque = deque[:len(deque)-1] // 队尾出队
        }

        // 3. 当前元素索引入队
        deque = append(deque, i)

        // 4. 当窗口形成（i &gt;= k-1），记录队首元素（当前窗口最大值）
        if i &gt;= k-1 {
            res = append(res, nums[deque[0]])
        }
    }

    return res
}
```

```Go
func topKFrequent(nums []int, k int) []int {
    map_num:=map[int]int{}
    //记录每个元素出现的次数
    for _,item:=range nums{
        map_num[item]++
    }
    h:=&amp;IHeap{}
    heap.Init(h)
    //所有元素入堆，堆的长度为k
    for key,value:=range map_num{
        heap.Push(h,[2]int{key,value})
        if h.Len()&gt;k{
            heap.Pop(h)
        }
    }
    res:=make([]int,k)
    //按顺序返回堆中的元素
    for i:=0;i&lt;k;i++{
        res[k-i-1]=heap.Pop(h).([2]int)[0]
    }
    return res
}

func (h IHeap) Len()int {

}

    h[i],h[j]=h[j],h[i]
}

    n:=len(old)
```

```Go
// 前序遍历：【处理中间、遍历左边、遍历右边】 栈：处理中间、入栈右边、入栈左边（出栈顺序同）
// 处理、访问左、访问右
// 访问是访问一个
func PreOrder2(root *TreeNode, res *[]int) {
    if root == nil {
       return
    }

    stack := []*TreeNode{}
    stack = append(stack, root)
    for len(stack) &gt; 0 {
       // 先处理根节点
       // 出栈
       node := stack[len(stack)-1]
       stack = stack[:len(stack)-1]
       left := node.Left
       right := node.Right
       *res = append(*res, node.Val)
       // 再加入左右节点 先加右节点 再加左节点 要求输出：根左右 ，出栈顺序 根左右
       if right != nil {
          stack = append(stack, right)
       }
       if left != nil {
          stack = append(stack, left)
       }
    }
    // 全部处理完了 返回结果
}

// 中序遍历：【遍历左边、处理中间、遍历右边】 栈：处理中间、入栈右边、入栈左边（出栈顺序同）
// 遍历左，处理中，访问右
// 遍历是全部
func InOrder2(root *TreeNode, res *[]int) {
    if root == nil {
       return
    }
    stack := make([]*TreeNode, 0)
    cur := root // 用cur追踪当前遍历节点，避免直接操作root

    // 核心循环：栈不为空 或 当前节点不为空时继续
    for len(stack) &gt; 0 || cur != nil {
       // 第一步：遍历到当前节点的最左子节点，全部入栈
       for cur != nil {
          stack = append(stack, cur)
          cur = cur.Left // 一直向左走
       }

       // 第二步：弹出栈顶节点（此时左子树已遍历完），处理该节点（根）
       node := stack[len(stack)-1]
       stack = stack[:len(stack)-1]
       *res = append(*res, node.Val)   // 处理

       // 第三步：处理当前节点的右子树（右）
       cur = node.Right
    }
}

// 后序遍历：左右根
// 遍历左，遍历右，处理中
// 反转就是根右左 ，前序遍历是根左右，稍微修改就是根右左
// 所以 后序遍历 是 前序遍历的修改后的反转
func PostOrder2(root *TreeNode, res *[]int) {
    if root == nil {
       return
    }

    stack := make([]*TreeNode, 0)
    stack = append(stack, root)
    for len(stack) &gt; 0 {
       node := stack[len(stack)-1]
       stack = stack[:len(stack)-1]
       *res = append(*res, node.Val)
       if node.Left != nil {
          stack = append(stack, node.Left)
       }
       if node.Right != nil {
          stack = append(stack, node.Right)
       }
    }
    *res = reverse(res)
}

func reverse(res *[]int) []int {
    for i, j := 0, len(*res)-1; i &lt; j; i, j = i+1, j-1 {
       (*res)[i], (*res)[j] = (*res)[j], (*res)[i]
    }
    return *res
}
```

当我们处理中序和后序的时候，发现：访问顺序和处理顺序不同，可能会出现下面的情况：

> Eg: 1 2 3 4 5 作为树（层序放）
> 
> 

1. 中序遍历：左根右。遍历到最左边，取栈中的节点（此时有1 2 4），取出4，发现左边没有，输出4，右边也没有节点，就继续下一个。下一个2，发现左边有节点（注意：4就是左节点，已经处理过了）加入4（这里出现问题了）。最简单的办法就是设置一个访问数组。（标记已访问的左子树）

注意： continue // continue 为了保证先把左子树全部处理完；visited 为了记录已经访问过的左子树节点

```Go
func InOrder3(root *TreeNode, res *[]int) {
    if root == nil {
       return
    }
    visited := make(map[*TreeNode]bool) // 标记节点是否已处理（避免重复处理）
    stack := []*TreeNode{root}          // 初始化栈，放入根节点

    for len(stack) &gt; 0 {
       // 取栈顶节点（不弹出）
       node := stack[len(stack)-1]
       // 1. 若左子节点存在且未入栈，优先处理左子树（入栈）
       if node.Left != nil &amp;&amp; !visited[node.Left] {
          stack = append(stack, node.Left)
          visited[node.Left] = true // 标记为已入栈（避免重复入栈）
          continue // continue 为了保证先把左子树全部处理完
       }

       // 2. 左子树已处理，弹出当前节点并记录（处理根）
       stack = stack[:len(stack)-1]
       *res = append(*res, node.Val)

       // 3. 处理右子树（入栈右节点）
       if node.Right != nil {
          stack = append(stack, node.Right)
          // 右节点无需提前标记，因为弹出父节点后才会处理右节点，不会重复
       }
    }
}
```

1. 后序遍历：左右根。先遍历左边，再遍历右边，最后访问根节点。同样的，先遍历左边到最后1 2 4，然后左右都没了，停止遍历。从栈中处理，先拿到4，左右都没，收集4\.再拿2，左右都有（此时左边已经处理过了，要标记），右边有，右边5入栈，不处理。处理5，收集5\.又到了2，左右都有（此时左右都处理了，所以两边都要标记）全部标记

```Go
func PostOrder3(root *TreeNode, res *[]int) {
    if root == nil {
       return
    }

    stack := make([]*TreeNode, 0)
    stack = append(stack, root)
    visited := make(map[*TreeNode]bool)

    for len(stack) &gt; 0 {
       node := stack[len(stack)-1]
       if node.Left != nil &amp;&amp; !visited[node.Left] {
          stack = append(stack, node.Left)
          visited[node.Left] = true
          continue
       }
       if node.Right != nil &amp;&amp; !visited[node.Right] {
          stack = append(stack, node.Right)
          visited[node.Right] = true
          continue
       }
       *res = append(*res, node.Val)
       stack = stack[:len(stack)-1]
    }
}
```

```Go
        dfs(node.Left, depth+1)
        // 再遍历右子树
        dfs(node.Right, depth+1)
    }

    // 根节点深度从 1 开始（或 0，只要统一即可）
    dfs(root, 1)
    return leftVal
```

```Go
func buildTree(inorder []int, postorder []int) *TreeNode {
    // 记录索引和数值
    im := make(map[int]int)
    for i, v := range inorder{
        im[v] = i
    }
    n := len(inorder)

    // 后序数组中从后向前找第一个数字为根节点
    // 将根节点在中序数组中寻找，找到对应的索引位置
    // 在索引的两侧分别为左子树全部节点和右子树全部节点
    // 左边的中序数组作为左子树的中序数组
    // 右边的中序数组作为右子树的中序数组
    // 具有相同数字的后序数组作为左右子树的后序数组(递归)
    if len(postorder) == 0{
        // 防止空数组导致下面出现数组访问越界
        return nil
    }
    r := postorder[n-1]
    rnode := Node(r)
    index, _ := im[r]
    leftin := inorder[:index]
    rightin := inorder[index+1:]
    leftpost := postorder[:index]
    rightpost := postorder[index:n-1] 

    // 左子树递归
    leftNode := buildTree(leftin, leftpost)
    // 右子树递归
    rightNode := buildTree(rightin, rightpost)
    rnode.Left = leftNode
    rnode.Right = rightNode
    return rnode
}
```

```Go
        stack = stack[:len(stack)-1]

        if pre != nil {
            if current.Val == pre.Val {
                count++
            } else {
                // 结算前一个元素
                updateRes(pre.Val, &amp;count, &amp;maxCount, &amp;res)
                count = 1
            }
        }
        pre = current

        // 3. 处理右子树
        current = current.Right
    }

    // 结算最后一个元素
    updateRes(pre.Val, &amp;count, &amp;maxCount, &amp;res)

    return res
}

// 辅助函数：根据当前计数更新结果集(这里指针为了改变上面的数字、数组)
// 要是不清楚，可以直接替换到上面的代码中
func updateRes(val int, count, maxCount *int, res *[]int) {
    if *count &gt; *maxCount {
        // 当前元素出现次数更多，替换结果集
        *maxCount = *count
        *res = []int{val}
    } else if *count == *maxCount {
        // 当前元素出现次数与最大值相同，追加到结果集
        *res = append(*res, val)
    }
    // 次数更少则不处理
}
```

```Go
/**
 * Definition for a binary tree node.
 * type TreeNode struct {
 *     Val int
 *     Left *TreeNode
 *     Right *TreeNode
 * }
 */
func deleteNode(root *TreeNode, key int) *TreeNode {
    if root == nil {
        return nil
    }
    // 去右子树删除（key比当前节点大）
    if key &gt; root.Val {
        root.Right = deleteNode(root.Right, key)
    } else if key &lt; root.Val { // 去左子树删除（key比当前节点小）
        root.Left = deleteNode(root.Left, key)
    } else { // 当前节点就是要删除的节点，分3种情况处理
        // 情况1：无左子树，直接返回右子树（右子树为空则返回nil，等价删除）
        if root.Left == nil {
            return root.Right
        }
        // 情况2：无右子树，直接返回左子树
        if root.Right == nil {
            return root.Left
        }
        // 情况3：左右子树都存在
        // 找右子树的最左节点（右子树的最小值节点）
        node := root.Right
        for node.Left != nil {
            node = node.Left
        }
        // 将原节点的左子树，挂载到右子树最左节点的左子树
        node.Left = root.Left
        // 用右子树顶替原节点位置（原节点被删除）
        root = root.Right
    }
    return root
}
```

## 修剪二叉搜索树

```Go
//  这里直接利用二叉搜索树的大小规律。节点比较之后就可以知道两侧的大小结果了
func trimBST(root *TreeNode, low int, high int) *TreeNode {
    if root == nil{
        return nil
    }

    if root.Val &lt; low {
        // 当前节点比最小值小 → 左子树全小，直接返回修剪后的右子树（右子树可能有符合条件的）
        return trimBST(root.Right, low, high)
    }
    if root.Val &gt; high {
        // 当前节点比最大值大 → 右子树全大，直接返回修剪后的左子树（左子树可能有符合条件的）
        return trimBST(root.Left, low, high)
    }

    root.Left = trimBST(root.Left, low, high)
    root.Right = trimBST(root.Right, low, high)

    return root
}
```

```Go
import "fmt"

func letterCombinations(digits string) []string {
    // 边界条件：空字符串直接返回空
    if digits == "" {
        return []string{}
    }

    path, res := "", make([]string, 0)
    // 数字到字母的映射（0/1无对应字母，直接留空）
    cm := map[int]string{
        0: "",
        1: "",
        2: "abc",
        3: "def",
        4: "ghi",
        5: "jkl",
        6: "mno",
        7: "pqrs",
        8: "tuv",
        9: "wxyz",
    }

    // 转换digits为数字数组（只允许2-9）
    nums, err := getNums(digits)
    if err != nil {
        return []string{}
    }

    var dfs func(index int)
    dfs = func(index int) {
        // 终止条件：路径长度 == 数字长度（所有数字都选完字母）
        if len(path) == len(nums) {
            res = append(res, path)
            return
        }

        // 当前数字对应的字母集合
        number := nums[index]
        chs := cm[number]
        // 遍历当前数字的所有字母（横向遍历）
        for _, ch := range chs {
            // 选择当前字母
            path = fmt.Sprintf("%s%c", path, ch)
            // 递归处理下一个数字（纵向遍历）
            dfs(index + 1)
            // 回溯：撤销当前选择（关键！否则path会一直累加）
            if len(path) &gt; 0 {
                path = path[:len(path)-1]
            }
        }
    }

    dfs(0)
    return res
}

// getNums：将digits字符串转为数字数组（仅允许2-9）
func getNums(s string) ([]int, error) {
    nums := make([]int, 0, len(s))
    for _, c := range s {
        // 只允许数字2-9，0/1/非数字均视为无效
        if c &lt; '2' || c &gt; '9' {
            return nil, fmt.Errorf("invalid character: %c", c)
        }
        nums = append(nums, int(c-'0'))
    }
    return nums, nil
}
```

## 组合总和

和单纯的组合问题相比:

1. 相同:

1. 和单纯的组合问题同样的有一个目标集合\[1\-n\],这里是\[给定数组\]

2. 避免重复,同样存在横向剪枝的情况

3. 同样需要 sum 进行剪枝\(和组合总和III的问题相比\)

4. startIndex的作用同样是用来取数的\(这里也是startIndex作为下标的开始\)

```Go
func combinationSum(candidates []int, target int) [][]int {
    path, res := make([]int, 0), make([][]int, 0)
    sum := 0

    var dfs func(startIndex int)
    dfs = func(startIndex int) {
        // 终止条件：sum超过target也终止（剪枝）
        if sum &gt; target {
            return
        }
        // 找到目标和，加入结果集
        if sum == target {
            temp := make([]int, len(path)) // 按path长度初始化，避免copy无效
            copy(temp, path)
            res = append(res, temp)
            return
        }

        // 横向遍历：从startIndex开始，避免重复组合（如[2,3]和[3,2]）
        for i := startIndex; i &lt; len(candidates); i++ {
            // 剪枝：当前sum+候选值&gt;target，后续候选值更大（需先排序，下面说明），直接break(可选)
            if sum+candidates[i] &gt; target {
                break
            }
            // 剪单个值,作用不大(跳过某个值)(可选)
            // if sum+candidates[i] &gt; target {
            //     continue
            // }
            // 选择当前候选值
            sum += candidates[i]
            path = append(path, candidates[i])
            // 递归：允许重复选当前值（所以传i，不是i+1！）
            dfs(i)
            // 回溯：撤销选择（sum必须同步回溯，否则累加错误）
            path = path[:len(path)-1]
            sum -= candidates[i]
        }
```

```Go
func combinationSum2(candidates []int, target int) [][]int {
    path, res := make([]int, 0), make([][]int, 0)

    dfs = func(startIndex int) {

            return
        }
        // 找到目标和，加入结果集
        if sum == target {
            temp := make([]int, len(path)) // 按path长度初始化，避免copy无效
            copy(temp, path)
            res = append(res, temp)
            return
        }

        // 横向遍历：从startIndex开始，避免重复组合（如[2,3]和[3,2]）
        for i := startIndex; i &lt; len(candidates); i++ {
            // 剪枝1：当前sum+候选值&gt;target，后续候选值更大（需先排序，下面说明），直接break(可选)
            if sum+candidates[i] &gt; target {
                break
            }
            // 剪枝2: 去掉重复数字开头
            if i &gt; startIndex &amp;&amp; candidates[i] == candidates[i-1]{
                continue
            }
            // 选择当前候选值
            sum += candidates[i]
            path = append(path, candidates[i])
            // 递归：不允许重复选当前值
            dfs(i+1)
            // 回溯：撤销选择（sum必须同步回溯，否则累加错误）
            path = path[:len(path)-1]
            sum -= candidates[i]
        }
    }

    // (必要)
    // 1. 关键：排序后才能让剪枝（sum+candidates[i]&gt;target时break）生效
    // 2. 关键: 排序后同一层才能去重
    sort.Ints(candidates)
    dfs(0)

    return res
}
```

```Go
func partition(s string) [][]string {
    path, res := make([]string, 0), make([][]string, 0)
    n := len(s)

    // dfs 函数：从 index 开始分割字符串
    var dfs func(index int)
    dfs = func(index int) {
        // 终止条件：index 到达字符串末尾，说明分割完成，收集结果
        if index == n {
            // 复制 path，避免后续修改影响结果
            temp := make([]string, len(path))
            copy(temp, path)
            res = append(res, temp)
            return
        }

        // 横向遍历：枚举所有以 index 为起点的子串（index ~ i）
        for i := index; i &lt; n; i++ {
            // 剪枝：如果当前子串不是回文，直接跳过（无需递归）
            if !help(s, index, i) {
                continue
            }
            // 选择当前子串
            path = append(path, s[index:i+1])
            // 递归：下一次分割从 i+1 开始
            dfs(i + 1)
            // 回溯：撤销选择
            path = path[:len(path)-1]
        }
    }

    dfs(0)
    return res
}

func help(s string, left, right int) bool {
    for left &lt; right {
        if s[left] != s[right] {
            return false
        }
        left++
        right--
    }
    return true
}
```

## \[注意\]break和continue一定要分清

\<callout emoji="📍"\>

在回溯算法的剪枝中，`break` 和 `continue` 的核心区别是 **“是否终止当前层级的后续遍历”**——`continue` 是 “跳过当前元素，继续遍历同一层级的下一个元素”，`break` 是 “直接终止当前层级的所有遍历，不再看后续元素”。选择哪一个，完全取决于 **“当前元素不合法时，后续元素是否还有可能合法”**。

```Go
func permute(nums []int) [][]int {
    var res [][]int          // 结果集
    n := len(nums)
    used := make([]bool, n)  // 标记元素是否已使用
    path := make([]int, 0, n) // 记录当前路径

    // 回溯函数
    var backtrack func()
    backtrack = func() {
        // 终止条件：路径长度等于数组长度，收集结果
        if len(path) == n {
            // 复制 path（切片是引用类型，需深拷贝避免后续修改影响结果）
            temp := make([]int, n)
            copy(temp, path)
            res = append(res, temp)
            return
        }

        // 横向遍历：枚举所有未使用的元素
        for i := 0; i &lt; n; i++ {
            // 跳过已使用的元素（剪枝）
            if used[i] {
                continue 
            }

            // 选择：标记元素为已使用，加入路径
            used[i] = true
            path = append(path, nums[i])

            // 递归：继续向下搜索下一个元素
            backtrack()

            // 回溯：撤销选择（移除路径最后一个元素，标记为未使用）
            path = path[:len(path)-1]
            used[i] = false
        }
    }

    backtrack()
    return res
}
```

```Go
import "sort"

func findItinerary(tickets [][]string) []string {
    // 1. 构建邻接表：from -&gt; 排序后的to列表（保证字典序）
    adj := make(map[string][]string)
    for _, ticket := range tickets {
        from, to := ticket[0], ticket[1]
        adj[from] = append(adj[from], to)
    }
    // 对每个from的to列表排序（确保优先选字典序小的）
    for key := range adj {
        sort.Strings(adj[key])
    }

    res := make([]string, 0)
    // 2. DFS回溯找欧拉路径
    var dfs func(from string)
    dfs = func(from string) {
        // 循环：当前from还有未使用的机票时，继续递归
        for len(adj[from]) &gt; 0 {
            // 贪心选字典序最小的to

            adj[from] = adj[from][1:]
            // 递归访问下一个节点
            dfs(to)
        }

        res = append(res, from)
```

```Go
func solveNQueens(n int) [][]string {
    res := make([][]string, 0)
    if n == 0 {
        return res
    }
    // 初始化棋盘：n行n列，初始值为 '.'
    chessboard := make([]string, n)
    for i := range chessboard {
        row := make([]byte, n)
        for j := range row {
            row[j] = '.'
        }
        chessboard[i] = string(row)
    }

    var dfs func(row int)
    dfs = func(row int) {
        // 终止条件：递归到最后一行
        if row == n {
            // 深拷贝当前棋盘状态到结果集
            temp := make([]string, n)
            copy(temp, chessboard)
            res = append(res, temp)
            return
        }

        // 遍历当前行的每一列，尝试放置皇后
        for col := 0; col &lt; n; col++ {
            // 剪枝：当前位置不合法，跳过
            if !isValid(row, col, chessboard) {
                continue
            }

            // 选择：在当前位置放置皇后（将 string 类型转换为 byte 数组修改后再重新赋值）
            rowBytes := []byte(chessboard[row])
            rowBytes[col] = 'Q'
            chessboard[row] = string(rowBytes)

            // 递归：处理下一行
            dfs(row + 1)

            // 回溯：撤销选择，恢复为 '.'
            rowBytes[col] = '.'
            chessboard[row] = string(rowBytes)
        }
    }

    dfs(0)
    return res
}

// isValid 检查 (row, col) 位置放置皇后是否合法
// 由于每一层需要检查的是已经放过的行（上面的斜线：左上和右上）
func isValid(row, col int, chessboard []string) bool {
    n := len(chessboard)

    // 检查同一列是否有皇后
    for i := 0; i &lt; row; i++ {
        if chessboard[i][col] == 'Q' {
            return false
        }
    }

    // 检查左上到右下的 45 度对角线
    for i, j := row-1, col-1; i &gt;= 0 &amp;&amp; j &gt;= 0; i, j = i-1, j-1 {
        if chessboard[i][j] == 'Q' {
            return false
        }
    }

    // 检查右上到左下的 135 度对角线
    for i, j := row-1, col+1; i &gt;= 0 &amp;&amp; j &lt; n; i, j = i-1, j+1 {
        if chessboard[i][j] == 'Q' {
            return false
        }
    }

    return true
}
```

```Go
func solveSudoku(board [][]byte) {
    var backtracking func(board [][]byte) bool
    backtracking = func(board [][]byte) bool {
        for i := 0; i &lt; 9; i++ {
            for j := 0; j &lt; 9; j++ {
                num := board[i][j]
                //判断此位置是否适合填数字

                        }
                        board[i][j] = '.'
                    }

            }

    }
```

## 接雨水

```Go
// 单调栈：其实这个是按照行算的。水量 = 宽 * 高 （加和的时机：出现一次凹槽）
// 暴力解法是按照列算的，是每个列
func trap(height []int) int {
   if len(height) &lt;= 2 {
      return 0 // 少于3列无法存水
   }
   // 单调栈：存储索引，保持高度单调递减（初始压入第一个索引）
   st := make([]int, 1, len(height)) 
   var res int

   for i := 1; i &lt; len(height); i++ {
      // 核心：循环判断当前高度 &gt; 栈顶高度（形成凹槽）
      for len(st) &gt; 0 &amp;&amp; height[i] &gt; height[st[len(st)-1]] {
         // 弹出栈顶（凹槽底部）
         top := st[len(st)-1]
         st = st[:len(st)-1]

         // 栈空则无左边界，无法存水
         if len(st) == 0 {
            break
         }

         // 正确公式：计算当前凹槽的雨水量
         leftIdx := st[len(st)-1] // 左边界索引
         h := min(height[i], height[leftIdx]) - height[top] // 存水高度
         w := i - leftIdx - 1                               // 存水宽度
         res += h * w
      }
      // 所有比当前高的栈顶都弹出后，压入当前索引（保持栈单调递减）
      st = append(st, i)
   }
   return res
}

func min(x, y int) int {
   if x &gt;= y {
      return y
   }
   return x
}
```

```Go
func trap(height []int) int {
    // 暴力：每一列的雨水之和(p = min(l, r) - curHight)
    res := 0
    var curHight, l, r int
    for i:=0;i&lt;len(height);i++{
        if i == 0 || i == len(height)-1{
            continue
        }
        curHight = height[i]
        // 找左边最大
        l, r = curHight, curHight
        for j:=i-1;j&gt;=0;j--{
            if height[j] &gt; l{
                l = height[j]
            }

        for j:=i+1;j&lt;len(height);j++{

        res += min(l, r) - curHight
    } 
    return res
}
```

```Go
func largestRectangleArea(heights []int) int {

    // 1. 相比于接雨水，这是求内部，不是求外部（不重要）
    // 2. 首尾元素也需要处理（所以可以加个哨兵0）
    newHeights := append([]int{0}, heights...)

    // 这样的话，我们处理（找左边/右边第一个比它小的值时就可以统一处理）
    stack := make([]int, 1, len(newHeights)) // 默认最左边的0入栈
    res := 0
    for i := 1; i &lt; len(newHeights); i++ {
        // 遍历每一列（真实的列[1:len(newHeight)-1]）
        // 一旦有小的就要记录（处理）
        for len(stack) &gt; 0 &amp;&amp; newHeights[i] &lt; newHeights[stack[len(stack)-1]] {
            // 待求的列
            topIndex := stack[len(stack)-1]
            stack = stack[:len(stack)-1]
            // 左边第一个小的索引
            leftIndex := stack[len(stack)-1]
            // 右边界
            rightIndex := i
            // 宽 = 右边界(rightIndex) - 左边界(leftIndex) - 1
            l := rightIndex - leftIndex - 1
            // 高 = 栈顶元素的高度
            h := newHeights[topIndex]
            // 本列的最大矩形(这里选择进行比较)
            temp := l * h
            if temp &gt; res {
                res = temp
            }
        }
        stack = append(stack, i)
    }
    return res
}
```

```Go
    minRightIndex := make([]int, size)
```

求某个数在数组中，左边/右边的第一个大/小的值

具体分类：

1. 右边的第一个大的值（遍历顺序从左往右）

如果当前遍历到的值\<栈顶元素，这就是你要的答案。对应求出的当前遍历到的值，是栈顶元素的右边的第一个小的值。

如果当前遍历到的值\>栈顶元素，这就是你要的答案。对应求出的栈顶元素，是当前元素的左边的第一个大的值。

题目：给定一个整数 n，求以 1 \.\.\. n 为节点组成的二叉搜索树有多少种？（数学归纳法）

N == 3时：

当1为头结点的时候，其右子树有两个节点，看这两个节点的布局，是不是和 n 为2的时候两棵树的布局是一样的啊！

当3为头结点的时候，其左子树有两个节点，看这两个节点的布局，是不是和n为2的时候两棵树的布局也是一样的啊！

当2为头结点的时候，其左右子树都只有一个节点，布局是不是和n为1的时候只有一棵树的布局也是一样的啊！

元素1为头结点搜索树的数量 = 右子树有2个元素的搜索树数量 \* 左子树有0个元素的搜索树数量

元素2为头结点搜索树的数量 = 右子树有1个元素的搜索树数量 \* 左子树有1个元素的搜索树数量

元素3为头结点搜索树的数量 = 右子树有0个元素的搜索树数量 \* 左子树有2个元素的搜索树数量

有2个元素的搜索树数量就是dp\[2\]。

有1个元素的搜索树数量就是dp\[1\]。

有0个元素的搜索树数量就是dp\[0\]。

元素n为头结点搜索树的数量 = 右子树有j\-1个元素的搜索树数量 \* 左子树有i\-j个元素的搜索树数量\(j\-1 从0开始到i\-1，那么 j 从1开始到i   每组的和\)

所以dp\[3\] = dp\[2\] \\* dp\[0\] \+ dp\[1\] \\* dp\[1\] \+ dp\[0\] \* dp\[2\]

那么：dp\[i\] = dp\[j\-1\]\*dp\[i\-j\]   \(左边有j\-1个，右边有i\-j个\) j从1\-i（各个元素分别为头节点）

```Go
func numTrees(n int) int {
    // dp:数量为i的二叉搜索树的种数
    // i:节点数量
    // j相当于是头结点的元素，从1遍历到i为止
    // 公式:dp[i] = dp[j-1]*dp[i-j] (左边有j-1个，右边有i-j个) j从1-i（各个元素分别为头节点）
    // 遍历顺序:从前往后
    // 初始化:dp[0]=1
    dp := make([]int, n+1)
    dp[0]=1
    // 推导:
    // 从 1 到 n 推导
    for i:=1;i&lt;=n;i++{
        for j:=1;j&lt;=i;j++{
            dp[i] += dp[j-1]*dp[i-j]
        }
    }
    return dp[n]
}
```

```Go
            }
        }
```

### \[重要\]01背包理论

问题背景：

有n件物品和一个最多能背重量为w 的背包。第i件物品的重量是weight\[i\]，得到的价值是value\[i\] 。每件物品只能用一次，求解将哪些物品装入背包里物品价值总和最大。（简单说，有限背包容量\[体积/重量\]，有限物品个数，物品具有体积/重量，求拿取物品的最大总价值）

问题具象化：

暴力想法：

每个物品只有两个状态，取或者不取。那么遍历所有的情况就是2的n次方。时间复杂度为O\(2^n\)，指数级别的时间复杂度，所以需要动态规划进行优化。

动态规划五部曲：

1. dp：求的是物品最大总价值。dp就表示该条件下的最大总价值（从下图中：dp\[i\]\[j\] 表示从下标为\[0\-i\]的物品里任意取，放进容量为j的背包，价值总和最大是多少。）

2. i，j：这里用二维dp。条件包括：背包容量和物品个数，有两个维度。（随意）i表示物品个数，j表示物体容量（最后发现：\[1,4\]表示物品0\-1在背包容量为4时，可以取得的最大总价值）

1. 递推公式：dp\[i\]\[j\]=max\(dp\[i \- 1\]\[j\], dp\[i \- 1\]\[j \- weight\[i\]\] \+ value\[i\]\)（核心：如果放物品x， 那么背包要先留出物品x的容量。也就是这个物品先放。因为它必放）\(不放，那就是前面一个同容量的情况相同\)放和不放体现在这个背包的容量是否变化！至于价值都要依赖前面一个的值！

2. 初始化：（i: 物品索引   j：背包容量）

1. 背包重量为0，什么都放不了：dp\[i\]\[0\]=0

2. 放第一个物品时，背包能放的最大价值：`if j>=weight[0] {dp[0][j]=value[0]}`

3. \[想偷懒！\]可以，我们发现，容量为0价值也是0；没初始化的也是0；那就只初始化拿第一个物品的总价值

4. 遍历顺序：这里二维dp嘛，要遍历背包容量和物品。其实都可以，为了好理解，和上面的理解相同。就选择遍历先遍历物品，再遍历背包容量。

5. 推导：尽量除去初始化的部分（少算/保证不出现数组越界）

物品从第二行遍历（i=1），背包容量从第二列遍历（j=1）。dp\[i\]\[j\]=max\(dp\[i \- 1\]\[j\], dp\[i \- 1\]\[j \- weight\[i\]\] \+ value\[i\]\)。

1. 返回：dp\[m\]\[n\]: m个物品，背包容量为n条件下的总价值。

### \[重要\]0\-1 背包问题的识别

**核心特征**：**每个元素 / 物品 “只能选一次”**（单次选择） \+ **子集需满足特定约束条件**（如重量、和、体积等维度的限制）；**问题类型细分**：根据约束的目标，分为两类 ——

1. **可行性问题**：判断是否存在满足约束的子集（如分割等和子集：是否存在和为 target 的子集）；

2. **优化型问题**：在满足约束的前提下，最大化 / 最小化某个指标（如经典 0\-1 背包：容量约束下最大化价值）。

**更通俗的类比**

0\-1 背包就像 “从一堆物品里挑子集，每个物品最多拿一次，且挑出来的子集要符合某个规矩”：

- “规矩” 可以是 “子集总重量不超过背包容量”（经典背包），也可以是 “子集和等于总和的一半”（分割等和）；

- 问题可以是 “有没有符合规矩的子集”（可行性），也可以是 “符合规矩的子集里，价值最大是多少”（优化型）。

## 深度优先搜索（模板题）\-\- 有向无环图

打印输出 1\-n 的所有路径

```Go
package main

import "fmt"

func main() {
    var n, m int
    fmt.Scan(&n, &m)

    // 邻接矩阵存储图（1~n 节点）
    graph := make([][]int, n+1)
    for i := range graph {
        graph[i] = make([]int, n+1)
    }

    // 读取边
    for i := 0; i < m; i++ {
        var from, to int
        fmt.Scan(&from, &to)
        graph[from][to] = 1 // 表示 from 到 to 有边
    }

    // 存储所有路径和当前路径
    res := make([][]int, 0)
    path := []int{1} // 起点 1 先加入路径

    // 深度优先搜索找所有路径
    var dfs func(x int)
    dfs = func(x int) {
        // 到达终点 n，记录路径
        if x == n {
            temp := append([]int{}, path...) // 拷贝当前路径
            res = append(res, temp)
            return
        }

        // 遍历所有可能的下一个节点（1~n）
        for i := 1; i <= n; i++ {
            // 存在 x 到 i 的边
            if graph[x][i] == 1 {
                // 处理：加入当前节点
                path = append(path, i)
                // 递归：访问下一个节点
                dfs(i)
                // 回溯：移除当前节点
                path = path[:len(path)-1]
            }
        }
    }

    // 从起点 1 开始搜索
    dfs(1)

    // 输出结果
    if len(res) == 0 {
        fmt.Print("-1")
        return
    }

    for _, path := range res {
        printPath(path)
    }
    return
}

func printPath(nums []int) {
    for i, v := range nums {
        if i == len(nums)-1 {
            fmt.Printf("%d", v)
        } else {
            fmt.Printf("%d ", v)
        }
    }
    fmt.Println()
}
```

## 深度优先搜索（模板题）\-\- 岛屿数量

注意：

1. 访问数组

2. 方向数组

3. 越界判断（越界直接跳过）

4. 遍历元素（特点：未访问 \+ 满足条件）

```Go
package main

import "fmt"

func main(){
    // 存储岛屿数据
    var n,m int
    fmt.Scan(&n,&m)
    g := make([][]int,n)
    for i := range g{
        g[i] = make([]int, m)
    }
    for i:=0;i<n;i++{
        for j:=0;j<m;j++{
            fmt.Scan(&g[i][j])
        }
    }
    // 定义访问数组
    visit := make([][]bool, n)
    for i:= range visit{
        visit[i] = make([]bool, m)
    }
    // 定义方向数组
    dir := [4][]int{
        []int{0, 1},
        []int{1, 0},
        []int{0, -1},
        []int{-1, 0},
    }

    // 计算岛屿的数量
    var dfs func(x, y int)
    dfs = func(x, y int){
        // 终止
        if visit[x][y] || g[x][y] == 0{
            return
        }
        // 处理
        visit[x][y] = true
        for i:=0;i<4;i++{
            // 确定移动方向
            newx := x + dir[i][0]
            newy := y + dir[i][1]
            // 越界判断
            if newx < 0 || newx >= len(g) || newy < 0 || newy >= len(g[0]){
                continue
            }
            // 递归
            dfs(newx, newy)
        }
    }

    res := 0

    for i:=0;i<len(g);i++{
        for j:=0;j<len(g[0]);j++{
            if !visit[i][j] && g[i][j] == 1{
                // 搜索每一个 没搜索过 + 陆地 节点
                res++
                dfs(i, j)
            }
        }
    }
    fmt.Print(res)
    return
}
```

```Go
    fmt.Scan(&n,&m)

        grid[i] = make([]int, m)

            fmt.Scan(&grid[i][j])

    visited := make([][]bool, n)
    for i := 0; i < n; i++ {
        visited[i] = make([]bool, m)
    }

    // 定义 BFS 闭包：捕获外部的 grid、visited、dir 变量
    bfs := func(x, y int) {
        // 初始化队列，存储坐标 [x, y]
        queue := [][]int{{x, y}}
        visited[x][y] = true // 加入队列后立即标记已访问

        // 队列不为空时循环
        for len(queue) > 0 {
            // 出队（取队首元素）
            cur := queue[0]
            queue = queue[1:] // 截取切片，模拟出队
            curx, cury := cur[0], cur[1]

            // 遍历四个方向
            for i := 0; i < 4; i++ {
                nextx := curx + dir[i][0]
                nexty := cury + dir[i][1]

                // 越界判断
                if nextx < 0 || nextx >= len(grid) || nexty < 0 || nexty >= len(grid[0]) {
                    continue
                }

                // 未访问且是陆地（值为 1），入队并标记
                if !visited[nextx][nexty] && grid[nextx][nexty] == 1 {
                    queue = append(queue, []int{nextx, nexty})
                    visited[nextx][nexty] = true
                }
            }
        }
    }

    // 统计岛屿数量
    result := 0
    for i := 0; i < n; i++ {
        for j := 0; j < m; j++ {
            if !visited[i][j] && grid[i][j] == 1 {
                result++
                bfs(i, j) // 调用闭包
            }
        }
    }

    // 输出结果
```

## 有向图是否有环（模板题）—— 有向有环/无环图

```Go
// 课程表：判断有向图是否存在【环】
// 思路：拓扑排序（BFS）
func canFinish(numCourses int, prerequisites [][]int) bool {
    // ---------------------
    // 1. 建图：有向图的【邻接表】
    // g[i] 存储：从节点 i 出发，能到达的所有邻接节点
    // 图论定义：有向边 u → v
    // ---------------------
    g := make([][]int, numCourses)

    // ---------------------
    // 2. 入度数组 in-degree
    // d[i]：节点 i 的【入度】→ 有多少条边指向 i
    // ---------------------
    d := make([]int, numCourses)

    // 遍历所有边，构建邻接表和入度数组
    for _, row := range prerequisites {
        course := row[0]       // 后继节点 v
        precourse := row[1]     // 前驱节点 u
        g[precourse] = append(g[precourse], course) // 添加有向边：u → v
        d[course]++ // 节点 v 的入度 +1
    }

    // ---------------------
    // 3. 初始化队列：放入所有【入度为 0】的节点
    // 拓扑排序起点：入度为0的节点（没有前驱，可以直接学）
    // ---------------------
    queue := []int{}
    for i, v := range d {
        if v == 0 {
            queue = append(queue, i)
        }
    }

    // ---------------------
    // 4. BFS 执行拓扑排序
    // 每次取出【入度为0】的节点，删除它的出边
    // ---------------------
    count := 0 // 统计拓扑序列的节点数量
    for len(queue) > 0 {
        // 取出队首节点 u（当前入度为0，可处理）
        u := queue[0]
        queue = queue[1:]
        count++ // 该节点加入拓扑序列

        // ---------------------
        // 遍历 u 的所有邻接节点 v（u → v）
        // 删掉边 u→v：v 的入度 -1
        // ---------------------
        for _, v := range g[u] {
            d[v]--
            // 如果 v 的入度变为 0，加入队列
            if d[v] == 0 {
                queue = append(queue, v)
            }
        }
    }

    // ---------------------
    // 5. 判断是否有环
    // 拓扑序列长度 == 总节点数 → 无环 → 可以完成课程
    // 拓扑序列长度 < 总节点数 → 有环 → 无法完成
    // ---------------------
    return count == numCourses
}
```

## 岛屿的最大面积

本质：

1. 找出每个岛屿\(每次遍历，都遍历完所有关联的节点，所以每次遍历后，就是一个岛屿\)

2. 计算每个岛屿的面积（累加）

3. DFS / BFS

```Go
// bfs 

package main

import "fmt"

func main() {
    // 输入
    var n, m int
    fmt.Scan(&n, &m)
    // 初始化
    g := make([][]int, n)
    for i := range g {
        g[i] = make([]int, m)
    }
    for i := 0; i < n; i++ {
        for j := 0; j < m; j++ {
            fmt.Scan(&g[i][j])
        }
    }

    // 处理
    res := 0 // 存储最大面积
    visited := make([][]bool, n)
    for i := range visited {
        visited[i] = make([]bool, m)
    }
    // 四个方向：右、下、上、左
    dir := [][]int{{0, 1}, {1, 0}, {-1, 0}, {0, -1}}

    // 定义 BFS 闭包，计算单个连通块的面积
    var bfs func(x, y int) int // 返回当前连通块的面积
    bfs = func(x, y int) int {
        temp := 0 // 每次 BFS 重置面积计数器
        queue := make([][]int, 0)
        queue = append(queue, []int{x, y})
        visited[x][y] = true
        temp++ // 初始节点计入面积

        for len(queue) > 0 {
            node := queue[0]
            queue = queue[1:]
            curx, cury := node[0], node[1] // 正确获取坐标

            // 遍历四个方向
            for i := 0; i < 4; i++ {
                nextx := curx + dir[i][0]
                nexty := cury + dir[i][1]

                // 越界判断
                if nextx < 0 || nextx >= len(g) || nexty < 0 || nexty >= len(g[0]) {
                    continue
                }

                // 未访问且是陆地，入队并标记
                if !visited[nextx][nexty] && g[nextx][nexty] == 1 {
                    queue = append(queue, []int{nextx, nexty})
                    visited[nextx][nexty] = true
                    temp++ // 新增节点计入面积
                }
            }
        }
        return temp // 返回当前连通块的面积
    }

    // 遍历网格，寻找最大连通块
    for i := 0; i < n; i++ {
        for j := 0; j < m; j++ {
            if g[i][j] == 1 && !visited[i][j] {
                currentArea := bfs(i, j)
                res = max(res, currentArea)
            }
        }
    }

    // 输出结果
    fmt.Println(res)
}

        return a
    }
```

正确版本：优化（读取优化 \+ 顺流而下）

```Go
package main

import (
    "bufio"
    "fmt"
    "os"
    "strconv"
    "strings"
)

var directions = [][]int{{0, -1}, {0, 1}, {-1, 0}, {1, 0}} // 四个方向的偏移量

func main() {
    scanner := bufio.NewScanner(os.Stdin)

    // 读取行列数
    scanner.Scan()
    lineList := strings.Fields(scanner.Text())
    N, _ := strconv.Atoi(lineList[0])
    M, _ := strconv.Atoi(lineList[1])

    // 读取矩阵
    grid := make([][]int, N)
    for i := 0; i < N; i++ {
        grid[i] = make([]int, M)
        scanner.Scan()
        lineList = strings.Fields(scanner.Text())
        for j := 0; j < M; j++ {
            grid[i][j], _ = strconv.Atoi(lineList[j])
        }
    }

    // 定义DFS闭包，直接访问外部的grid、directions、N、M，以及每次遍历的visited
    var dfs func(x, y int) (bool, bool)
    var visited [][]bool // 每次遍历新起点时重置

    // 闭包实现DFS逻辑
    dfs = func(x, y int) (bool, bool) {
        visited[x][y] = true
        // 第一组边界：i=0 或 j=0
        canReachFirst := (x == 0 || y == 0)
        // 第二组边界：i=N-1 或 j=M-1
        canReachSecond := (x == N-1 || y == M-1)

        // 遍历四个方向
        for _, dir := range directions {
            nextx := x + dir[0]
            nexty := y + dir[1]
            // 边界检查
            if nextx < 0 || nextx >= N || nexty < 0 || nexty >= M {
                continue
            }
            // 水流规则：只能流向更低或等高的位置，且未访问过
            if grid[nextx][nexty] <= grid[x][y] && !visited[nextx][nexty] {
                r1, r2 := dfs(nextx, nexty)
                // 合并子节点的遍历结果
                canReachFirst = canReachFirst || r1
                canReachSecond = canReachSecond || r2
            }
        }
        return canReachFirst, canReachSecond
    }

    // 遍历每个单元格，检查是否能同时到达两组边界
    for i := 0; i < N; i++ {
        for j := 0; j < M; j++ {
            // 每次DFS前重置visited数组
            visited = make([][]bool, N)
            for k := range visited {
```

```Bash
func main() {

        // 读取行列数
        scanner.Scan()
        lineList := strings.Fields(scanner.Text())
        N, _ := strconv.Atoi(lineList[0])
        M, _ := strconv.Atoi(lineList[1])

        // 读取矩阵
        grid := make([][]int, N)
        for i := 0; i < N; i++ {
                grid[i] = make([]int, M)
                scanner.Scan()
                lineList = strings.Fields(scanner.Text())
                for j := 0; j < M; j++ {
                        grid[i][j], _ = strconv.Atoi(lineList[j])
                }
        }

        // 方向数组：上下左右
        directions := [][]int{{0, -1}, {0, 1}, {-1, 0}, {1, 0}}

        // 定义两个标记数组，记录能从对应边界逆流而上到达的点
        reachFirst := make([][]bool, N)  // 第一组边界（i=0/j=0）可达
        reachSecond := make([][]bool, N) // 第二组边界（i=N-1/j=M-1）可达
        for i := range reachFirst {
                reachFirst[i] = make([]bool, M)
                reachSecond[i] = make([]bool, M)
        }

        // 定义BFS闭包（逆流而上：从边界向更高/等高的位置遍历）
        var bfs func(startPoints [][]int, visited [][]bool)
        bfs = func(startPoints [][]int, visited [][]bool) {
                queue := startPoints
                // 初始化起始边界点为已访问
                for _, p := range startPoints {
                        x, y := p[0], p[1]
                        visited[x][y] = true
                }

                // BFS核心逻辑
                for len(queue) > 0 {
                        curr := queue[0]
                        queue = queue[1:]
                        x, y := curr[0], curr[1]

                        // 遍历四个方向，逆流而上（新位置 >= 当前位置）
                        for _, dir := range directions {
                                nx, ny := x+dir[0], y+dir[1]
                                // 边界检查 + 未访问 + 高度条件（逆流）
                                if nx >= 0 && nx < N && ny >= 0 && ny < M && !visited[nx][ny] && grid[nx][ny] >= grid[x][y] {
                                        visited[nx][ny] = true
                                        queue = append(queue, []int{nx, ny})
                                }
                        }
                }
        }

        // 收集第一组边界点（i=0 或 j=0）并执行BFS
        startFirst := make([][]int, 0)
        for i := 0; i < N; i++ {
                for j := 0; j < M; j++ {
                        if i == 0 || j == 0 {
                                startFirst = append(startFirst, []int{i, j})
                        }
                }
        }
        bfs(startFirst, reachFirst)

        // 收集第二组边界点（i=N-1 或 j=M-1）并执行BFS
        startSecond := make([][]int, 0)
        for i := 0; i < N; i++ {
                for j := 0; j < M; j++ {
                        if i == N-1 || j == M-1 {
                                startSecond = append(startSecond, []int{i, j})
                        }
                }
        }
        bfs(startSecond, reachSecond)

        // 筛选两个标记的交集（能同时流向两组边界的点）
        result := make([][]int, 0)
        for i := 0; i < N; i++ {
                for j := 0; j < M; j++ {
                        if reachFirst[i][j] && reachSecond[i][j] {
                                result = append(result, []int{i, j})
                        }
                }
        }

        // 输出结果
        for _, pos := range result {
                fmt.Printf("%d %d\n", pos[0], pos[1])
        }
```

```Go
package main

import (
    "fmt"

func main() {
    var n, m int
    // 读取地图行数n、列数m
    fmt.Scan(&n, &m)
    // 初始化地图
    g := make([][]int, n)
    for i := range g {
        g[i] = make([]int, m)
    }
    // 读取地图数据（0=海水，1=陆地）
    for i := 0; i < n; i++ {
        for j := 0; j < m; j++ {
            fmt.Scan(&g[i][j])
        }
    }

    // 1. 第一步：标记所有岛屿并记录每个岛屿的面积
    visit := make([][]bool, n)
    for i := range visit {
        visit[i] = make([]bool, m)
    }
    islandArea := make(map[int]int) // 岛屿编号 -> 面积
    mark := 2                       // 岛屿编号从2开始（避免和0/1冲突）
    dir := [][]int{{0, 1}, {1, 0}, {0, -1}, {-1, 0}} // 上下左右四个方向
    isAllLand := true               // 标记是否全是陆地

    // DFS函数：标记岛屿、计算面积
    var dfs func(x, y int)
    dfs = func(x, y int) {
        // 边界判断：越界则返回
        if x < 0 || x >= n || y < 0 || y >= m {
            return
        }
        // 已访问 或 是海水：返回
        if visit[x][y] || g[x][y] != 1 {
            return
        }
        // 标记为已访问
        visit[x][y] = true
        // 标记当前格子属于哪个岛屿
        g[x][y] = mark
        // 岛屿面积+1（通过闭包更新）
        islandArea[mark]++

        // 遍历四个方向
        for _, d := range dir {
            nextX := x + d[0]
            nextY := y + d[1]
            dfs(nextX, nextY)
        }
    }

    // 遍历整个地图，标记所有岛屿
    for i := 0; i < n; i++ {
        for j := 0; j < m; j++ {
            // 存在海水，说明不是全陆地
            if g[i][j] == 0 {
                isAllLand = false
            }
            // 未访问的陆地：启动DFS标记岛屿
            if !visit[i][j] && g[i][j] == 1 {
                islandArea[mark] = 0 // 初始化当前岛屿面积
                dfs(i, j)            // 注意：原代码传了x/y（未定义），应传i/j
                mark++               // 下一个岛屿编号+1
            }
        }
    }

    // 特殊情况：全是陆地，最大面积就是地图总面积
    if isAllLand {
        fmt.Println(n * m)
        return
    }

    // 2. 第二步：遍历所有海水格子，计算添加陆地后的最大面积
    maxArea := 0
    for i := 0; i < n; i++ {
        for j := 0; j < m; j++ {
            // 只处理海水格子（添加陆地的位置）
            if g[i][j] == 0 {
                // 记录当前格子周围的岛屿编号（去重）
                visitedIslands := make(map[int]bool)
                currentArea := 1 // 新增的陆地本身占1
                // 遍历四个方向
                for _, d := range dir {
                    nearX := i + d[0]
                    nearY := j + d[1]
                    // 边界判断
                    if nearX < 0 || nearX >= n || nearY < 0 || nearY >= m {
                        continue
                    }
                    // 周围是岛屿（编号>=2）且未统计过
                    islandID := g[nearX][nearY]
                    if islandID >= 2 && !visitedIslands[islandID] {
                        currentArea += islandArea[islandID]
                        visitedIslands[islandID] = true // 标记已统计，避免重复加
                    }
                }
                // 更新最大面积
                if currentArea > maxArea {
                    maxArea = currentArea
                }
            }
        }
    }

    // 输出结果
    fmt.Println(maxArea)
}
```

### 四、计算岛屿总周长（仅DFS/BFS思路说明，周长统计无需递归/队列，逐格更高效）

> 说明：岛屿周长的核心是「统计每个陆地的四方向是否为海水/越界」，逐格判断是最优解（无需DFS/BFS），但如果非要用DFS/BFS实现，逻辑如下：

```Go
// DFS计算岛屿周长
func islandPerimeterDFS(grid [][]int) int {
        if len(grid) == 0 || len(grid[0]) == 0 {
                return 0
        }
        n, m := len(grid), len(grid[0])
        perimeter := 0

        var dfs func(x, y int)
        dfs = func(x, y int) {
                if x < 0 || x >= n || y < 0 || y >= m || grid[x][y] != 1 {
                        return
                }
                grid[x][y] = 0 // 标记已访问
                // 统计当前格子的四方向周长
                for _, d := range dirs {
                        nx, ny := x+d[0], y+d[1]
                        if nx < 0 || nx >= n || ny < 0 || ny >= m || grid[nx][ny] == 0 {
                                perimeter++
                        }
                }
                // 遍历四方向
                for _, d := range dirs {
                        dfs(x+d[0], y+d[1])
                }
        }

        // 遍历所有格子
        for i := 0; i < n; i++ {
                for j := 0; j < m; j++ {
                        if grid[i][j] == 1 {
                                dfs(i, j)
                        }
                }
        }
        return perimeter
}

// BFS计算岛屿周长（切片队列）
func islandPerimeterBFS(grid [][]int) int {
        if len(grid) == 0 || len(grid[0]) == 0 {
                return 0
        }
        n, m := len(grid), len(grid[0])
        perimeter := 0

        bfs := func(startX, startY int) {
                queue := [][]int{{startX, startY}}
                grid[startX][startY] = 0

                for len(queue) > 0 {
                        cur := queue[0]
                        queue = queue[1:]
                        x, y := cur[0], cur[1]

                        // 统计当前格子的四方向周长
                        for _, d := range dirs {
                                nx, ny := x+d[0], y+d[1]
                                if nx < 0 || nx >= n || ny < 0 || ny >= m || grid[nx][ny] == 0 {
                                        perimeter++
                                }
                        }

                        // 遍历四方向，入队未访问的陆地
                        for _, d := range dirs {
                                nx, ny := x+d[0], y+d[1]
                                if nx >= 0 && nx < n && ny >= 0 && ny < m && grid[nx][ny] == 1 {
                                        queue = append(queue, []int{nx, ny})
                                        grid[nx][ny] = 0
                                }
                        }
                }
        }

        // 遍历所有格子
        for i := 0; i < n; i++ {
                for j := 0; j < m; j++ {
                        if grid[i][j] == 1 {
                                bfs(i, j)
                        }
                }
        }
        return perimeter
}
```

### 核心说明（切片模拟队列的关键）

1. **队列初始化**：`queue := [][]int{{startX, startY}}`（存入起始坐标）；

2. **队首出队**：`cur := queue[0]; queue = queue[1:]`（切片截取，舍弃第一个元素）；

3. **队尾入队**：`queue = append(queue, []int{nx, ny})`（追加新坐标到队列末尾）；

4. **队列终止条件**：`len(queue) > 0`（队列空则停止）。

这种方式无需依赖`container/list`包，纯切片实现队列，更轻量且符合Go的极简风格。

### 算法选型建议

- **DFS**：代码更简洁（递归天然适配连通块遍历），但递归深度过大会导致栈溢出（适合小地图）；

- **BFS**：用循环\+队列实现，无栈溢出风险（适合大地图），切片模拟队列的性能和标准队列几乎无差异。

```Go
    return nums

// 快排序
func QUICK(arr []int, low, high int) {
    if low < high { // 基线条件：子数组长度>1才需要排序（否则直接返回）
       pivotIndex := partition(arr, low, high) // 找基准值的正确位置
       QUICK(arr, low, pivotIndex-1)           // 递归排基准左侧（都≤基准）
       QUICK(arr, pivotIndex+1, high)          // 递归排基准右侧（都>基准）
    }
}

func partition(arr []int, low, high int) int {
    pivot := arr[low]
    l, r := low, high
    for l < r {
       // 左边：找到第一个大的
       for l < r && arr[l] <= pivot {
          l++
       }
       // 右边：找到第一个小的
       for l < r && arr[r] >= pivot {
          r--
       }
       // 交换
       arr[l], arr[r] = arr[r], arr[l]
    }
    // 将基准放到中间
    arr[low], arr[l] = arr[l], arr[low]
    return l
}
```

```Go
func sort1_range(nums []int, left, right int) []int {
    // 改动1：修正递归终止条件（核心！）
    if left == right {
        return []int{nums[left]} // 区间只有1个元素，直接返回
    }
    mid := left + (right - left)/2
    // 左边、右边
    l := sort1_range(nums, left, mid)
    r := sort1_range(nums, mid+1, right)
    return sort1_merge(l, r)
}

// 改动2：重写merge函数，实现完整的有序数组合并

    i, j := 0, 0

    for i < len(nums1) && j < len(nums2) {
        if nums1[i] < nums2[j] {
            res = append(res, nums1[i])
            i++
        } else {
            res = append(res, nums2[j])
            j++
```

## 堆排序

```Go
package main

import "fmt"

// 原地排序
// 时间复杂度 O(nlogn)
// 空间复杂度 O(logn) heapfy的递归栈占用

// 堆排序入口
func heapSort(arr []int) {
    n := len(arr)

    // 1. 构建最大堆（从最后一个非叶子节点开始）：目的构建一个最大堆
    // [4, 2, 5, 6, 2, 7]
    // n/2 - 1 是数学推导的结果，原始应该是：((n-1)-1)/2 = 最后一个节点的父节点 = 最后一个非叶子节点的索引
    // 基于 父节点索引 = (n-1)/2 计算得出的
    for i := n/2 - 1; i >= 0; i-- {
       heapify(arr, n, i)
    }

    // 2. 一个个从堆顶取出元素：目的排序（从堆顶取出最大值和末尾交换再下滤前n-1个数字，只剩最后一个数字）
    for i := n - 1; i > 0; i-- {
       // 当前堆顶最大，交换到数组末尾
       arr[0], arr[i] = arr[i], arr[0]
       // 调整剩余元素，重新成为最大堆
       heapify(arr, i, 0)
    }
}

// ===================== 核心：堆化（下沉）=====================
// 将以 i 为根的子树调整为最大堆
func heapify(arr []int, n int, i int) {
    largest := i     // 最大值先设为根节点
    left := 2*i + 1  // 左孩子
    right := 2*i + 2 // 右孩子

    // 如果左孩子更大
    if left < n && arr[left] > arr[largest] {
       largest = left
    }
    // 如果右孩子更大
    if right < n && arr[right] > arr[largest] {
       largest = right
    }

    // 如果最大值不是根节点，交换 + 继续下沉
    if largest != i {
       arr[i], arr[largest] = arr[largest], arr[i]
       heapify(arr, n, largest) // 递归堆化受影响的子树
    }
}

// 迭代版
// 迭代版堆化（无递归，真正 O(1) 空间）
func heapify(arr []int, n int, i int) {
    for {
        largest := i
        left := 2*i + 1
        right := 2*i + 2

        if left < n && arr[left] > arr[largest] {
            largest = left
        }
        if right < n && arr[right] > arr[largest] {
            largest = right
        }

        if largest == i {
            break // 不需要下沉了，退出
        }

        // 交换
        arr[i], arr[largest] = arr[largest], arr[i]
        i = largest // 继续往下沉，不用递归
    }
}

func main() {
    arr := []int{4, 2, 5, 6, 2, 7}
    heapSort(arr)
    fmt.Println("排序后：", arr)
}
```

## 哲学家问题

**哲学家问题**

---

阻塞原因：哲学家全都拿着同方向的一个叉子，导致所有哲学家都不能正常吃饭，从而导致死锁

---

解决办法：

1. 改变拿叉子的顺序 —— 奇偶哲学家（奇数先拿左边，后拿右边；偶数先拿右边，再拿左边）\[这是从阻塞的直接现象发现的方法\]

2. 原子性的分配 / 一次性分配足够的叉子 —— 通过互斥锁先加锁，等到拿到足够的叉子后解锁 \[从代码角度发现的方法\] —— 一个全局锁，并发度低

3. 减少并发抢夺的数量 —— 比如一共有 n 个人抢夺，限制同时申请叉子的哲学家数量 ≤ n\-1，那么一定至少有一个人可以吃饭，其他人等待释放后再抢 —— 并发度较高

---

并发度比较：；理论上，1 \> 3 \> 2（奇偶分序理论是 n 个同时进餐；限制并发是最大 n \- 1个；lock 是 1 个）

```Go
        defer wg.Done()

        // 【核心逻辑】奇偶分序拿叉子
        if p.name%2 == 0 {

                p.pLeft()

                forks[p.rightForkIdx].Lock()
                p.pRight()
                forks[p.leftForkIdx].Lock()
                p.pLeft()

                forks[p.rightForkIdx].Unlock()
                p.vRight()
        }
}
```

## LRU缓存

1. get：如果存在，拿出来并更新到最前面（MoveToFront\(\)）；不存在就返回\-1

2. put：如果存在，直接覆盖；如果不存在，先更新到最前面（PushFront\(\)），然后判断是否超出cap，超出则淘汰最近最久未使用的元素 Back\(\)

双向链表实现（内置）

```Go
// list的元素是一个 kv 对的原因：为了 O(1)通过 v 找到 k，从而操作
type entry struct {
    key, value int
}

type LRUCache struct {
    capacity  int
    list      *list.List // 双向链表
    keyToNode map[int]*list.Element
}

func Constructor(capacity int) LRUCache {
    return LRUCache{capacity, list.New(), map[int]*list.Element{}}
}

func (c *LRUCache) Get(key int) int {
    node := c.keyToNode[key]
    if node == nil { // 没有这本书
        return -1
    }
    c.list.MoveToFront(node) // 把这本书放在最上面
    return node.Value.(entry).value
}

func (c *LRUCache) Put(key, value int) {
    if node := c.keyToNode[key]; node != nil { // 有这本书
        node.Value = entry{key, value} // 更新
        c.list.MoveToFront(node) // 把这本书放在最上面
        return
    }
    c.keyToNode[key] = c.list.PushFront(entry{key, value}) // 新书，放在最上面
    if len(c.keyToNode) > c.capacity { // 书太多了
        // delete(c.keyToNode, c.list.Remove(c.list.Back()).(entry).key) // 去掉最后一本书
        deleteKey := c.list.Remove(c.list.Back()).(entry).key
        delete(c.keyToNode, deleteKey)
    }
}
```

```Go
        return nil
    }
    // 分治
    return mergeDive(lists, 0, len(lists)-1)
}

func mergeDive(lists []*ListNode, left, right int) *ListNode{
    if left == right{
        // 同一个链表，说明只剩下一个了
        return lists[left]
    }
    mid := left + (right - left)/2
    // 分别合并左半部分、右半部分
    l1 := mergeDive(lists, left, mid)
    l2 := mergeDive(lists, mid+1, right)
    // 最后合并两部分
    return mergeList(l1, l2)
}
```

```Go
 * Definition for singly-linked list.
 * type ListNode struct {
 *     Val int
 *     Next *ListNode
 * }
 */
func mergeInBetween(list1 *ListNode, a int, b int, list2 *ListNode) *ListNode {
    // 找 a前驱 b后继
    l := list1
    for i:=0;i<a-1;i++{
        l = l.Next
    }
    r := l
    for i:=0;i<b-a+2;i++{
        // 从 a-1 到 b+1
        r = r.Next
    }
    // 接list2
    if list2 == nil{
        l.Next = r
        return list1
    }
    l.Next = list2
    end := list2
    for end.Next != nil{
        end = end.Next
    }
    end.Next = r
    return list1
}
```

## LFU缓存

```Go
type entry struct {
    key, value, freq int // freq 表示这本书被看的次数
}

type LFUCache struct {
    capacity   int
    minFreq    int
    keyToNode  map[int]*list.Element
    freqToList map[int]*list.List
}

func Constructor(capacity int) LFUCache {
    return LFUCache{
        capacity:   capacity,
        keyToNode:  map[int]*list.Element{},
        freqToList: map[int]*list.List{},
    }
}

func (c *LFUCache) pushFront(e *entry) {
    if _, ok := c.freqToList[e.freq]; !ok {
        c.freqToList[e.freq] = list.New() // 双向链表
    }
    c.keyToNode[e.key] = c.freqToList[e.freq].PushFront(e)
}

func (c *LFUCache) getEntry(key int) *entry {
    node := c.keyToNode[key]
    if node == nil { // 没有这本书
        return nil
    }
    e := node.Value.(*entry)
    lst := c.freqToList[e.freq]
    lst.Remove(node) // 把这本书抽出来
    if lst.Len() == 0 { // 抽出来后，这摞书是空的
        delete(c.freqToList, e.freq) // 移除空链表
        if c.minFreq == e.freq { // 这摞书是最左边的
            c.minFreq++
        }
    }
    e.freq++ // 看书次数 +1
    c.pushFront(e) // 放在右边这摞书的最上面
    return e
}

func (c *LFUCache) Get(key int) int {
    if e := c.getEntry(key); e != nil { // 有这本书
        return e.value
    }
    return -1 // 没有这本书
}

func (c *LFUCache) Put(key, value int) {
    if e := c.getEntry(key); e != nil { // 有这本书
        e.value = value // 更新 value
        return
    }
    if len(c.keyToNode) == c.capacity { // 书太多了
        lst := c.freqToList[c.minFreq] // 最左边那摞书
        k := lst.Remove(lst.Back()).(*entry).key // 移除最左边那摞书的最下面的书
        delete(c.keyToNode, k)
        if lst.Len() == 0 { // 这摞书是空的
            delete(c.freqToList, c.minFreq) // 移除空链表
        }
    }
    c.pushFront(&entry{key, value, 1}) // 新书放在「看过 1 次」的最上面
    c.minFreq = 1
}
```
